import mongoose from 'mongoose'
import cors from 'cors'
import multer from 'multer'

const app = express();
const port = 5000;

// Middleware
app.use(cors());
app.use(express.json());

  useNewUrlParser: true,

const formSchema = new mongoose.Schema({
  lastName: String,
  firstName: String,
  middleName: String,
  position: String,
  branch: String,
  address: String,
  mobileNo: String,
  whatsappNo: String,
  email: String,
  dateOfAppointment: Date,
  permanentDate: Date,
  dateOfBirth: Date,
  dateOfRetirement: Date,
  bankRegNo: String,
  bankBranch: String,
  oldUserID: String,  // Optional
  oldSubscriptionAmt: String,  // Optional
  hrmsNo: String,
  nominee1: String,
  relation1: String,
  nominee2: String,
  relation2: String,
  signature: Buffer,
});

const Form = mongoose.model('Form', formSchema);

// Multer setup for file uploads
const storage = multer.memoryStorage();
const upload = multer({ storage: storage });

// Routes
app.post('/api/form', upload.single('signature'), async (req, res) => {
  try {
    const formData = new Form({
      ...req.body,
      signature: req.file.buffer,
    });
    await formData.save();
    res.status(201).send('Form data saved successfully');
  } catch (error) {
    res.status(400).send('Error saving form data');
  }
});

// Get all forms
app.get('/api/forms', async (req, res) => {
  try {
    const forms = await Form.find();
    res.status(200).json(forms);
  } catch (error) {
    res.status(500).send('Error fetching form data');
  }
});

app.listen(port, () => console.log(`Server running on port ${port}`));
